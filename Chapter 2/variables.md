# variables.py
# Variable is a name given to a memory location that stores data. 
# In Python, variables are created when you assign a value to them. 
# You can use variables to store different types of data, such as numbers, strings, lists, and more.

# Data types in Python:
a = 10  # Integer variable
b = 3.14  # Float variable
c = "Hello, World!"  # String variable
d = [1, 2, 3]  # List variable
e = True  # Boolean variable
f = None  # NoneType variable

# operators in Python:
# Arithmetic operators: +, -, *, /, %, ** : power, // : round float value
# assignment operators: =, +=, -=, *=, /=, %=, **=, //=
# logical operators: and, or, not
# comparison operators: ==, !=, >, <, >=, <=
